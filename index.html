SCORE
0

0 km/h
`; const container = document.getElementById('b3d-container'); let renderer, scene, camera, playerBike, road, traffic = [], trees = [], buildings = [], lights = []; let clock, keys = { left: false, right: false, accel: false, brake: false, nitro: false }; let state = { speed: 0, maxSpeed: 42, nitro: 0, score: 0, distance: 0, over: false, lean: 0, playerX: 0 }; let raf = 0, destroyed = false, overlayEl = null; const ROAD_WIDTH = 14; const LANE_POS = [-4.2, 0, 4.2]; // ---------- Materials ---------- const matRoad = new THREE.MeshStandardMaterial({ color: 0x1a1228, roughness: 0.85, metalness: 0.15 }); const matMark = new THREE.MeshBasicMaterial({ color: 0xeef0ff }); const matRumble = new THREE.MeshStandardMaterial({ color: 0xff2e7e, emissive: 0x330011, roughness: 0.7 }); const matGrass = new THREE.MeshStandardMaterial({ color: 0x0a1f0a, roughness: 0.95 }); const matBikeBody = new THREE.MeshStandardMaterial({ color: 0x00c8ff, metalness: 0.75, roughness: 0.25, emissive: 0x001122 }); const matCarBody = [ new THREE.MeshStandardMaterial({ color: 0xff2e7e, metalness: 0.6, roughness: 0.3 }), new THREE.MeshStandardMaterial({ color: 0x7dff6a, metalness: 0.6, roughness: 0.3 }), new THREE.MeshStandardMaterial({ color: 0xffe14d, metalness: 0.6, roughness: 0.3 }), new THREE.MeshStandardMaterial({ color: 0xb026ff, metalness: 0.6, roughness: 0.3 }), new THREE.MeshStandardMaterial({ color: 0xff7a1a, metalness: 0.6, roughness: 0.3 }) ]; function createMotorcycle() { const g = new THREE.Group(); // Main body const body = new THREE.Mesh(new THREE.BoxGeometry(0.55, 0.32, 1.5), matBikeBody); body.position.y = 0.55; body.castShadow = true; g.add(body); // Fuel tank / fairing const tank = new THREE.Mesh(new THREE.BoxGeometry(0.48, 0.28, 0.7), matBikeBody); tank.position.set(0, 0.72, 0.15); tank.castShadow = true; g.add(tank); // Seat const seat = new THREE.Mesh(new THREE.BoxGeometry(0.4, 0.12, 0.55), matBikeDark); seat.position.set(0, 0.68, -0.35); g.add(seat); // Front fork const fork = new THREE.Mesh(new THREE.BoxGeometry(0.12, 0.55, 0.12), matBikeDark); fork.position.set(0, 0.55, 0.85); g.add(fork); // Handlebars const bars = new THREE.Mesh(new THREE.BoxGeometry(0.7, 0.06, 0.08), matBikeDark); bars.position.set(0, 0.85, 0.82); g.add(bars); // Headlight const head = new THREE.Mesh(new THREE.SphereGeometry(0.12, 12, 12), matLight); head.position.set(0, 0.62, 1.05); g.add(head); const headGlow = new THREE.PointLight(0xffffee, 1.2, 18); headGlow.position.copy(head.position); g.add(headGlow); // Rear brake light const brakeL = new THREE.Mesh(new THREE.BoxGeometry(0.35, 0.08, 0.06), matBrake); brakeL.position.set(0, 0.55, -0.78); g.add(brakeL); // Wheels function makeWheel(z) { const wg = new THREE.Group(); const tire = new THREE.Mesh(new THREE.TorusGeometry(0.28, 0.09, 10, 18), matTire); tire.rotation.y = Math.PI / 2; wg.add(tire); const rim = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.18, 0.08, 12), matWheel); rim.rotation.z = Math.PI / 2; wg.add(rim); wg.position.set(0, 0.28, z); return wg; } g.add(makeWheel(0.72)); g.add(makeWheel(-0.62)); // Exhaust const exh = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.05, 0.6, 8), matBikeDark); exh.rotation.z = Math.PI / 2; exh.position.set(0.28, 0.38, -0.3); g.add(exh); g.scale.set(1.15, 1.15, 1.15); return g; } function createCar(colorIdx) { const g = new THREE.Group(); const body = new THREE.Mesh(new THREE.BoxGeometry(1.6, 0.55, 3.2), matCarBody[colorIdx % matCarBody.length]); body.position.y = 0.55; body.castShadow = true; g.add(body); const cabin = new THREE.Mesh(new THREE.BoxGeometry(1.4, 0.45, 1.5), matBikeDark); cabin.position.set(0, 1.0, -0.2); g.add(cabin); // simple wheels [[-0.7, 1.1], [0.7, 1.1], [-0.7, -1.1], [0.7, -1.1]].forEach(([x, z]) => { const w = new THREE.Mesh(new THREE.CylinderGeometry(0.28, 0.28, 0.18, 10), matTire); w.rotation.z = Math.PI / 2; w.position.set(x, 0.28, z); g.add(w); }); return g; } function createTree() { const g = new THREE.Group(); const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.25, 1.8, 6), matTrunk); trunk.position.y = 0.9; g.add(trunk); const foliage = new THREE.Mesh(new THREE.ConeGeometry(1.1, 2.4, 7), matTree); foliage.position.y = 2.6; foliage.castShadow = true; g.add(foliage); return g; } function createBuilding(h) { const g = new THREE.Group(); const w = 3 + Math.random() * 3; const d = 3 + Math.random() * 2.5; const body = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), matBuilding); body.position.y = h / 2; body.castShadow = true; body.receiveShadow = true; g.add(body); // windows const winMat = matWindow; for (let y = 1.2; y < h - 0.8; y += 1.1) { for (let x = -w / 2 + 0.5; x < w / 2 - 0.4; x += 0.9) { if (Math.random() > 0.35) { const win = new THREE.Mesh(new THREE.PlaneGeometry(0.35, 0.45), winMat); win.position.set(x, y, d / 2 + 0.01); g.add(win); } } } return g; } function initThree() { const w = container.clientWidth || 640; const h = container.clientHeight || 480; renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false }); renderer.setSize(w, h); renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2)); renderer.shadowMap.enabled = true; renderer.shadowMap.type = THREE.PCFSoftShadowMap; renderer.setClearColor(0x05030d); container.appendChild(renderer.domElement); scene = new THREE.Scene(); scene.fog = new THREE.FogExp2(0x12082a, 0.012); camera = new THREE.PerspectiveCamera(60, w / h, 0.1, 400); camera.position.set(0, 3.2, 8); // Lighting const amb = new THREE.AmbientLight(0x334466, 0.55); scene.add(amb); const sun = new THREE.DirectionalLight(0xffe0c0, 1.15); sun.position.set(30, 60, 20); sun.castShadow = true; sun.shadow.mapSize.set(1024, 1024); sun.shadow.camera.near = 10; sun.shadow.camera.far = 120; sun.shadow.camera.left = -40; sun.shadow.camera.right = 40; sun.shadow.camera.top = 40; sun.shadow.camera.bottom = -40; scene.add(sun); // Road const roadGeo = new THREE.PlaneGeometry(ROAD_WIDTH + 4, 600); road = new THREE.Mesh(roadGeo, matRoad); road.rotation.x = -Math.PI / 2; road.position.z = -250; road.receiveShadow = true; scene.add(road); // Center & lane markings for (let i = 0; i < 80; i++) { const m = new THREE.Mesh(new THREE.PlaneGeometry(0.25, 3.5), matMark); m.rotation.x = -Math.PI / 2; m.position.set(0, 0.02, -i * 8); scene.add(m); } // Rumble strips [[-ROAD_WIDTH / 2 - 0.4, -ROAD_WIDTH / 2 + 0.15], [ROAD_WIDTH / 2 - 0.15, ROAD_WIDTH / 2 + 0.4]].forEach(([x1, x2]) => { for (let i = 0; i < 60; i++) { const r = new THREE.Mesh(new THREE.PlaneGeometry(0.55, 2.2), matRumble); r.rotation.x = -Math.PI / 2; r.position.set((x1 + x2) / 2, 0.03, -i * 10); scene.add(r); } }); // Grass shoulders const grassL = new THREE.Mesh(new THREE.PlaneGeometry(40, 600), matGrass); grassL.rotation.x = -Math.PI / 2; grassL.position.set(-ROAD_WIDTH / 2 - 22, -0.05, -250); scene.add(grassL); const grassR = grassL.clone(); grassR.position.x = ROAD_WIDTH / 2 + 22; scene.add(grassR); // Player bike playerBike = createMotorcycle(); playerBike.position.set(0, 0, 0); scene.add(playerBike); // Trees for (let i = 0; i < 70; i++) { const t = createTree(); const side = Math.random() > 0.5 ? 1 : -1; t.position.set(side * (ROAD_WIDTH / 2 + 4 + Math.random() * 12), 0, -Math.random() * 500); scene.add(t); trees.push(t); } // Buildings for (let i = 0; i < 35; i++) { const b = createBuilding(6 + Math.random() * 14); const side = Math.random() > 0.5 ? 1 : -1; b.position.set(side * (ROAD_WIDTH / 2 + 10 + Math.random() * 8), 0, -Math.random() * 480); scene.add(b); buildings.push(b); } // Street lights for (let i = 0; i < 25; i++) { const post = new THREE.Mesh(new THREE.CylinderGeometry(0.08, 0.1, 5.5, 6), matBikeDark); post.position.set((i % 2 === 0 ? -1 : 1) * (ROAD_WIDTH / 2 + 1.8), 2.75, -i * 20); scene.add(post); const lamp = new THREE.PointLight(0xffeeaa, 1.4, 22); lamp.position.set(post.position.x, 5.4, post.position.z); scene.add(lamp); lights.push({ post, lamp }); } // Traffic for (let i = 0; i < 12; i++) { const car = createCar(i); car.position.set(LANE_POS[i % 3], 0, -40 - i * 28 - Math.random() * 40); car.userData = { speed: 12 + Math.random() * 10, lane: i % 3 }; scene.add(car); traffic.push(car); } clock = new THREE.Clock(); } function reset() { state.speed = 0; state.nitro = 30; state.score = 0; state.distance = 0; state.over = false; state.lean = 0; state.playerX = 0; if (playerBike) { playerBike.position.set(0, 0, 0); playerBike.rotation.set(0, 0, 0); } traffic.forEach((c, i) => { c.position.z = -50 - i * 30 - Math.random() * 50; c.position.x = LANE_POS[i % 3]; }); if (overlayEl) { overlayEl.remove(); overlayEl = null; } updateHUD(); } function updateHUD() { const scoreEl = document.getElementById('bScore'); const speedEl = document.getElementById('bSpeed'); const nitroFill = document.getElementById('bNitroFill'); if (scoreEl) scoreEl.textContent = Math.floor(state.score); if (speedEl) speedEl.innerHTML = Math.floor(state.speed * 9.5) + ' km/h'; if (nitroFill) nitroFill.style.width = Math.min(100, state.nitro) + '%'; } function showGameOver() { if (overlayEl) return; const best = (window.state && window.state.highScores && window.state.highScores[id]) || 0; const div = document.createElement('div'); div.innerHTML = `
CRASHED!
Score ${Math.floor(state.score)} · Distance ${Math.floor(state.distance)} m
Best: ${best}
`; overlayEl = div.firstElementChild; document.querySelector('#stage .game-stage').appendChild(overlayEl); document.getElementById('bRestart').onclick = () => { reset(); if (!destroyed) raf = requestAnimationFrame(loop); }; // save score if (window.state && typeof window.saveScore === 'function') { window.saveScore(id, Math.floor(state.score)); } else if (window.state && window.state.highScores) { window.state.highScores[id] = Math.max(window.state.highScores[id] || 0, Math.floor(state.score)); } } function loop() { if (destroyed) return; raf = requestAnimationFrame(loop); const dt = Math.min(0.033, clock.getDelta()); if (!state.over) { // Input let steer = 0; if (keys.left) steer -= 1; if (keys.right) steer += 1; // Acceleration / braking / nitro let accel = 0; if (keys.accel) accel += 18; if (keys.brake) accel -= 28; if (keys.nitro && state.nitro > 0) { accel += 35; state.nitro = Math.max(0, state.nitro - 25 * dt); state.score += 40 * dt; } else { state.nitro = Math.min(100, state.nitro + 4 * dt); } if (!keys.accel && !keys.brake && !keys.nitro) accel -= 4; // natural drag state.speed = Math.max(0, Math.min(state.maxSpeed + (keys.nitro ? 18 : 0), state.speed + accel * dt)); state.distance += state.speed * dt; state.score += state.speed * 1.8 * dt; // Steering state.playerX += steer * (8 + state.speed * 0.35) * dt; state.playerX = Math.max(-ROAD_WIDTH / 2 + 1.2, Math.min(ROAD_WIDTH / 2 - 1.2, state.playerX)); state.lean += (steer * 0.45 - state.lean) * 8 * dt; // Move world toward player const move = state.speed * dt; traffic.forEach(c => { c.position.z += move - c.userData.speed * dt; if (c.position.z > 15) { c.position.z = -180 - Math.random() * 80; c.position.x = LANE_POS[Math.floor(Math.random() * 3)]; c.userData.speed = 10 + Math.random() * 14; } // simple AI lane change occasionally if (Math.random() < 0.002) { c.userData.lane = Math.floor(Math.random() * 3); } c.position.x += (LANE_POS[c.userData.lane] - c.position.x) * 1.5 * dt; }); trees.forEach(t => { t.position.z += move; if (t.position.z > 30) t.position.z -= 520; }); buildings.forEach(b => { b.position.z += move; if (b.position.z > 40) b.position.z -= 500; }); lights.forEach(l => { l.post.position.z += move; l.lamp.position.z += move; if (l.post.position.z > 25) { l.post.position.z -= 500; l.lamp.position.z -= 500; } }); road.position.z = -250 + (state.distance % 8); // Player transform playerBike.position.x = state.playerX; playerBike.rotation.z = -state.lean; playerBike.rotation.x = -state.speed * 0.004; // Camera follow camera.position.x += (state.playerX * 0.55 - camera.position.x) * 4 * dt; camera.position.y = 3.0 + state.speed * 0.02; camera.position.z = 7.5 - state.speed * 0.04; camera.lookAt(state.playerX * 0.3, 0.8, -6); // Collision for (const c of traffic) { const dx = c.position.x - playerBike.position.x; const dz = c.position.z - playerBike.position.z; if (Math.abs(dx) < 1.5 && Math.abs(dz) < 2.4) { state.over = true; showGameOver(); break; } // near miss score if (Math.abs(dx) < 2.8 && Math.abs(dz) < 4 && Math.abs(dz) > 2.4) { state.score += 15 * dt; } } // Collect nitro “bolts” (simple proximity reward when near traffic edges) if (Math.random() < 0.01 && state.speed > 15) { state.nitro = Math.min(100, state.nitro + 8); state.score += 25; } updateHUD(); } renderer.render(scene, camera); } // Input handlers function onKeyDown(e) { if (state.over && (e.code === 'Space' || e.code === 'Enter')) { reset(); return; } if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = true; if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = true; if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.accel = true; if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.brake = true; if (e.code === 'ShiftLeft' || e.code === 'ShiftRight' || e.code === 'KeyN') keys.nitro = true; } function onKeyUp(e) { if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = false; if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = false; if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.accel = false; if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.brake = false; if (e.code === 'ShiftLeft' || e.code === 'ShiftRight' || e.code === 'KeyN') keys.nitro = false; } function holdBtn(sel, setter) { const el = document.querySelector(sel); if (!el) return; const down = e => { e.preventDefault(); setter(true); }; const up = e => { e.preventDefault(); setter(false); }; el.addEventListener('pointerdown', down); el.addEventListener('pointerup', up); el.addEventListener('pointerleave', up); el.addEventListener('pointercancel', up); } // Start function startWhenReady() { if (typeof THREE === 'undefined') { setTimeout(startWhenReady, 50); return; } initThree(); reset(); window.addEventListener('keydown', onKeyDown); window.addEventListener('keyup', onKeyUp); holdBtn('#bLeft', v => keys.left = v); holdBtn('#bRight', v => keys.right = v); holdBtn('#bNitro', v => keys.nitro = v); // also allow holding accelerate by default on mobile via nitro button proximity, but keep simple keys.accel = true; // auto-accelerate for better mobile feel (common in endless racers) const onResize = () => { if (!renderer || !camera) return; const w = container.clientWidth; const h = container.clientHeight; camera.aspect = w / h; camera.updateProjectionMatrix(); renderer.setSize(w, h); }; window.addEventListener('resize', onResize); raf = requestAnimationFrame(loop); } startWhenReady(); return { destroy() { destroyed = true; cancelAnimationFrame(raf); window.removeEventListener('keydown', onKeyDown); window.removeEventListener('keyup', onKeyUp); if (renderer) { renderer.dispose(); if (renderer.domElement && renderer.domElement.parentNode) { renderer.domElement.parentNode.removeChild(renderer.domElement); } } if (overlayEl) overlayEl.remove(); } }; }
